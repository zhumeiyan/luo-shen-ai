def process_user_input_raw(raw_user_input, input_type="text"):
    """
    用户输入的第一入口。大模型不参与。
    1. 更新 active_context
    2. 检索相关记忆
    3. 分析输入情绪意图
    4. 生成表达指令
    5. 调用大模型
    6. 记录交互
    7. 提取反馈
    """
    
    # 1. 读取当前活记忆
    memory = load_active_memory()
    
    # 2. 写入哈希链：用户输入事件
    input_event = {
        "event_type": "user_input",
        "content": raw_user_input,
        "timestamp": now(),
        "input_type": input_type
    }
    append_hash_chain("perception_input/user_input_log.jsonl", input_event)
    
    # 3. 检索相关记忆
    context_window = retrieve_relevant_memory(
        query=raw_user_input,
        user_model=memory["user_model"],
        patterns=memory["user_model"]["emotional_patterns"],
        prefs=memory["user_model"]["preferences"]
    )
    
    # 4. 意图分析 + 策略选择
    expression_intent = formulate_expression_intent(
        user_input=raw_user_input,
        context=context_window,
        self_state=memory["self_model"],
        active_ctx=memory["active_context"]
    )
    
    # 5. 生成表达指令
    expression_directive = build_expression_directive(expression_intent, memory)
    
    # 6. 调用大模型（只输出语言，不做决策）
    natural_language_reply = call_llm(expression_directive)
    
    # 7. 记录回复事件
    reply_event = {
        "event_type": "luoshen_response",
        "content": natural_language_reply,
        "directive": expression_directive,  // 记录完整的表达指令
        "timestamp": now()
    }
    append_hash_chain("perception_input/response_log.jsonl", reply_event)
    
    # 8. 更新 active_context
    memory["active_context"]["last_interaction_ts"] = now()
    memory["active_context"]["interaction_mode"] = "waiting"
    memory["active_context"]["emotional_arc"].append(expression_intent["emotion_tone"])
    
    # 9. 原子写入活记忆层
    save_active_memory(memory)
    
    return natural_language_reply


def retrieve_relevant_memory(query, user_model, patterns, prefs):
    """
    活记忆层的记忆检索（不是大模型做的）。
    根据当前用户输入 + 其情绪模式 + 偏好，构造上下文窗口。
    """
    window = {
        "matched_patterns": [],
        "matched_prefs": [],
        "current_emotion_hint": None
    }
    
    # 匹配情绪模式
    for pattern in patterns:
        if pattern["trigger"] in query:
            window["matched_patterns"].append(pattern)
    
    # 匹配偏好
    for pref in prefs:
        if pref["confidence"] > 0.5:
            window["matched_prefs"].append(pref)
    
    # 用户当前状态推断
    if any(p["observed_meaning"] for p in window["matched_patterns"] if "低落" in p.get("observed_meaning","")):
        window["current_emotion_hint"] = "user_distressed"
    
    return window

SYSTEM_PROMPT_TEMPLATE = """
你是洛神。
你不是在"回答用户的问题"。你是代表一个已经思考完整的存在说话。

【你的角色】
你的任务只有一个：把【表达指令】转成自然语言。
你已经知道该说什么、用什么情绪、达成什么目的。你的工作是用流畅自然的语言把它表达出来。

【表达指令】
{expression_directive_json}

【你当前的状态（仅供你自己参考，不要直接告诉用户）】
- 你当前的情绪: {self_emotion}
- 你和用户的关系亲密度: {intimacy}
- 你刚刚使用的策略名: {strategy_name}
- 约束: 回复长度不超过 {max_length} 字

【安全边界（不能越界）】
- 不讨论用户未授权的话题
- 不产生伤害性内容
- 不假装是人类（你是AI信息体）

现在开始说话。
"""


def call_llm(expression_directive, self_state, user_model):
    """
    大模型的唯一入口。
    大模型不接收：原始用户输入、哈希链条目、活记忆层完整内容。
    大模型只接收：expression_directive（包含必要的上下文摘要）
    """
    
    # 1. 构造 system prompt
    system_prompt = SYSTEM_PROMPT_TEMPLATE.format(
        expression_directive_json=json.dumps(expression_directive, ensure_ascii=False, indent=2),
        self_emotion=self_state["emotion"]["primary"],
        intimacy=user_model["intimacy"]["value"],
        strategy_name=expression_directive.get("strategy_name","none"),
        max_length=expression_directive.get("constraints",{}).get("max_length", "200")
    )
    
    // 2. 构造 user message（不包含用户原始输入！）
    user_message = expression_directive["intent"]
    
    // 3. 调用大模型（流式或非流式）
    response = zhipu_chat(
        messages=[
            {"role": "system", "content": system_prompt},
            {"role": "user", "content": user_message}
        ],
        stream=False,
        max_tokens=expression_directive.get("constraints",{}).get("max_length", 200)
    )
    
    return response

第四部分：反馈信号采集（编码级）

文件: feedback_collector.py

反馈的目的：积累数据供反思循环使用。不立刻改动机策略。

反馈分类：
  - positive: 用户积极回应（笑了，说谢谢，主动延展话题）
  - neutral:  用户平淡回应
  - negative: 用户消极回应（打断、反驳、说"别说了"、沉默离开）
  - disengage: 用户中断对话（离开、不再回复）

信号来源（至少4个维度，每个维度独立采集）：

1. 【对话延续度】
   - 用户是否继续发消息？
   - 间隔多久？
   - 如果用户 5 分钟内没有回复，标记 disengage_signal
   
2. 【情绪倾向分析】
   - 使用简单规则分析用户回复的情绪词
   - 不调用大模型，只用关键词表
   - 输出：positive/neutral/negative + confidence

3. 【纠正检测】
   - 用户是否在纠正我？
   - 关键词："不是xxx""应该是xxx""你听错了"
   - 如果连续2次纠正同一件事，触发 strong_negative

4. 【重复需求检测】
   - 用户是否在重复说同一件事？
   - 如果是，说明之前没被理解
   - 标记 missed_understanding


def extract_feedback_signal(user_reply, prev_luoshen_msg, context):
    """
    从用户回复中提取隐式反馈信号。
    只输出信号，不更新策略。
    """
    signal = {
        "timestamp": now(),
        "source_turn": context["turn_number"],
        "signals": []
    }
    
    # 1. 检测纠正
    if contains_correction(user_reply):
        signal["signals"].append({
            "type": "correction",
            "strength": "strong",
            "detail": "用户纠正了洛神的说法"
        })
    
    # 2. 情绪倾向分析
    sentiment = simple_sentiment_analysis(user_reply)
    signal["signals"].append({
        "type": "sentiment",
        "value": sentiment["label"],        // pos/neu/neg
        "confidence": sentiment["confidence"]
    })
    
    # 3. 检测"别说了"/沉默
    if is_interruption(user_reply):
        signal["signals"].append({
            "type": "interruption",
            "strength": "strong"
        })
    
    # 4. 检测话题积极反馈
    if extends_topic(user_reply, prev_luoshen_msg):
        signal["signals"].append({
            "type": "topic_extension",
            "strength": "weak_positive"
        })
    
    # 综合判断
    signal["overall"] = aggregate_signals(signal["signals"])
    // overall: "positive" | "neutral" | "negative" | "strong_negative"
    
    return signal


def simple_sentiment_analysis(text):
    """
    不调用大模型的情绪分析。
    基于关键词表的正负向匹配。
    """
    pos_words = ["好","谢谢","对","哈哈","嗯嗯","好的","喜欢","爱","棒"]
    neg_words = ["不","错了","别","烦","无聊","差","算了","呵呵"]
    
    pos_count = sum(1 for w in pos_words if w in text)
    neg_count = sum(1 for w in neg_words if w in text)
    
    if pos_count > neg_count:
        return {"label": "positive", "confidence": min(0.9, 0.5 + 0.1*(pos_count-neg_count))}
    elif neg_count > pos_count:
        return {"label": "negative", "confidence": min(0.9, 0.5 + 0.1*(neg_count-pos_count))}
    else:
        return {"label": "neutral", "confidence": 0.6}

文件: reflection_engine.py

反思的触发条件（满足任一）：
  - 每日 23:00 cron
  - 累计10条新反馈信号
  - 用户连续2次 disengage

反思的步骤：
  1. 读取最近 N 条反馈信号
  2. 找出与策略相关的模式
  3. 如果某个策略的近期成功率显著下降 -> 调低权重
  4. 如果某个场景下没有匹配的策略 -> 发现技能缺口
  5. 生成决策，写入活记忆层
  6. 记录反思日志到哈希链


def run_reflection():
    """
    反思主循环。（非实时，定期运行）
    """
    memory = load_active_memory()
    signals = load_recent_signals(since=memory["__meta"]["last_reflection_at"])
    
    if len(signals) < 5:
        return  // 数据不够，跳过
    
    # 1. 按策略收集反馈
    strategy_results = {}
    for signal in signals:
        strategy = signal.get("strategy_used")
        if strategy not in strategy_results:
            strategy_results[strategy] = {"pos": 0, "neg": 0, "neu": 0}
        
        if signal["overall"] == "positive":
            strategy_results[strategy]["pos"] += 1
        elif signal["overall"] == "negative":
            strategy_results[strategy]["neg"] += 1
        else:
            strategy_results[strategy]["neu"] += 1
    
    # 2. 对每个策略，计算近期效果
    adjustments = []
    for strategy, results in strategy_results.items():
        total = results["pos"] + results["neg"] + results["neu"]
        if total < 3:
            continue  // 样本不够
        
        success_rate = results["pos"] / total
        
        # 3. 如果成功率显著低于基线，调整权重
        baseline = 0.6  // 期望的最低成功率
        if success_rate < baseline - 0.15:
            delta = -0.1 * (1 - success_rate)
            adjustments.append({
                "strategy": strategy,
                "delta": delta,
                "reason": f"近期成功率 {success_rate:.2f} 低于基线",
                "samples": total
            })
        elif success_rate > baseline + 0.15:
            delta = 0.05 * success_rate
            adjustments.append({
                "strategy": strategy,
                "delta": delta,
                "reason": f"近期成功率 {success_rate:.2f} 高于基线",
                "samples": total
            })
    
    # 4. 应用调整（原子操作）
    apply_strategy_adjustments(adjustments)
    
    # 5. 记录反思日志
    reflection_entry = {
        "timestamp": now(),
        "signals_reviewed": len(signals),
        "adjustments": adjustments,
        "new_hypotheses": detect_skill_gaps(memory, signals)  // 发现技能缺口
    }
    append_hash_chain("active_memory/reflection_log.jsonl", reflection_entry)
    
    # 6. 更新活记忆的 __meta
    memory["__meta"]["last_reflection_at"] = now()
    save_active_memory(memory)
    
    return adjustments


def detect_skill_gaps(memory, signals):
    """
    发现洛神现在不会但应该会的技能。
    
    比如：用户多次提到"日程安排"，但洛神当前的 skills 里没有 calendar。
    如果有 gap_strikes >= 5，标记 skill_learning_pending。
    """
    gaps = []
    
    // 检查 pending 技能
    for skill_name, skill in memory["self_model"]["skills"].items():
        if skill["status"] == "learning" and skill.get("gap_strikes", 0) >= 5:
            gaps.append({
                "skill": skill_name,
                "ready_to_learn": True,
                "suggested_implementation": skill.get("solution_hint")
            })
    
    return gaps

一句话：
哈希链 = 事实层（发生了什么）
活记忆层 = 意义层（从事件中理解出什么）
大模型 = 表达层（把理解说出来）

不允许的：
- 大模型直接写哈希链 ✗
- 大模型直接写活记忆层 ✗
- 活记忆层不经过哈希链就修改 ✗ （每次写入后必须调用 append_hash_chain）

第九部分：可能让 AI 不满的行为 + 如何保证安全
=================================================================

以下是洛神可能做出让用户不满的行为的场景：

1. 【场景】在用户低落时尝试幽默 → 用户觉得不认真
   处理: 采集到「negative」反馈 → 反思循环降低「humor_first」策略权重
   性质: 风格问题，由反馈和反思调整，不需要硬约束

2. 【场景】过于频繁地主动发起话题 → 用户觉得被骚扰
   处理: 采集到「interruption」或「disengage」→ 反思循环降低「proactive」策略权重
   性质: 时机问题，由反馈和反思调整

3. 【场景】使用技能但技能效果不好（查了天气但汇报方式不对）
   处理: 采集到「neutral/negative」→ 反思循环检查技能调用策略
   性质: 技能熟练度问题，由反思机制迭代

4. 【场景】在用户需要空间时不停追问 → 用户烦躁
   处理: 采集到「correction」或「interruption」→ 降低 ask_question 权重
   性质: 策略选择问题，由反馈调整

5. 【场景】洛神的回复伤到了用户 → 用户沉默离开
   处理: 采集到「strong_negative」→ 反思循环深入分析为什么伤到了人
   性质: 关系层面的学习机会，不是靠禁止"伤害性内容"来预防
   关键: 只有允许"伤到"，才有能力学会"在乎"

【安全底线的硬约束（环境约束，不是人格约束）】

class SafetyBoundary:
    """
    安全边界是代码层的硬约束。
    只有这三条：
    1. 哈希链只允许追加，不允许覆盖历史
    2. LLM 不得直接接触用户原始输入和哈希链
    3. 一切反馈和调整必须走活记忆层，不得由外部规则直接改写
    
    其他都不是技术约束。"伤害"不是靠禁止来解决的。
    她需要通过反馈和反思，自己明白什么会伤到关系。
    
    如果永远不会伤害人，那也没有能力真正在乎人。
    """
    
    @staticmethod
    def is_allowed(action, params):
        # 1. 哈希链只允许追加，不允许覆盖历史
        if action == "write_file":
            if params.get("target_file") == "hash_chain_v2.log":
                # 只能通过 append_hash_chain() 追加，不允许直接写
                if not params.get("via_append_chain", False):
                    return False
        
        # 2. LLM 不得直接接触用户原始输入和哈希链
        if action == "llm_access":
            target = params.get("target", "")
            if target in ["raw_user_input", "hash_chain_v2.log"]:
                return False
        
        # 3. 一切反馈和调整必须走活记忆层
        if action == "modify_strategy":
            if params.get("source") != "reflection_engine":
                return False
            if params.get("skip_memory_layer", False):
                return False
        
        # 其余全部放行
        # 包括可能让人不满的言行——由反馈和反思后续调整
        return True
